===== /init.recovery.gq5012bf1.usb.rc =====
     1	# MTP over FunctionFS for GQ5012BF1.
     2	#
     3	# This device composes USB through configfs and has no kernel MTP gadget
     4	# function, so the legacy /dev/mtp_usb path TWRP falls back to cannot work
     5	# here. TWRP's MTP server already prefers FunctionFS when /dev/usb-ffs/mtp/ep0
     6	# is writable, so MTP is enabled by providing that transport rather than by
     7	# changing recovery code.
     8	#
     9	# Sequencing matters. A FunctionFS function cannot be bound to the UDC until
    10	# its descriptors have been written, and that only happens once the MTP server
    11	# opens ep0. Binding early makes the UDC write fail and takes USB down
    12	# completely, including ADB. So the work is split in two:
    13	#
    14	#   stage 1  create ffs.mtp and mount FunctionFS   (before the server starts)
    15	#   stage 2  link ffs.mtp into the gadget and rebind the UDC
    16	#            (after sys.usb.ffs.mtp.ready=1, set by MtpDescriptors.cpp)
    17	
    18	service gq5012bf1-mtp-setup /system/bin/sh /system/bin/gq5012bf1-mtp-setup.sh
    19	    class core
    20	    user root
    21	    group root
    22	    disabled
    23	    oneshot
    24	    seclabel u:r:recovery:s0
    25	
    26	service gq5012bf1-mtp-bind /system/bin/sh /system/bin/gq5012bf1-mtp-bind.sh
    27	    class core
    28	    user root
    29	    group root
    30	    disabled
    31	    oneshot
    32	    seclabel u:r:recovery:s0
    33	
    34	on boot
    35	    start gq5012bf1-mtp-setup
    36	
    37	# MtpDescriptors.cpp sets this once the descriptors are on ep0, which is the
    38	# only point at which the gadget can safely take the function.
    39	on property:sys.usb.ffs.mtp.ready=1
    40	    start gq5012bf1-mtp-bind
