fb_write() {
    local label=$1 target=$2; shift 2
    local command=${1:-} required_mode capacity image_bytes image_path
    [[ $target == scratch || $target == *_b ]] || fatal "forbidden non-B write target: $target"
    [[ $target != super && $target != super_a && $target != super_b && $target != *_a ]] || fatal "forbidden write target: $target"
    [[ $(raw_fastboot_value current-slot) == a ]] || fatal 'current slot changed away from A before write'
    case $target in
        scratch|system_b|system_ext_b|product_b|vendor_b|vendor_dlkm_b|system_dlkm_b|odm_dlkm_b) required_mode=fastbootd ;;
        boot_b|init_boot_b|vendor_boot_b|dtbo_b|vbmeta_b|vbmeta_system_b|vbmeta_vendor_b) required_mode=bootloader ;;
        *) fatal "unclassified write target: $target" ;;
    esac
    [[ $DEVICE_MODE == "$required_mode" ]] || fatal "$target write attempted in $DEVICE_MODE instead of $required_mode"
    if [[ $required_mode == fastbootd ]]; then
        [[ $(raw_fastboot_value is-userspace) == yes ]] || fatal 'userspace fastboot lost before write'
    else
        [[ $(raw_fastboot_value is-userspace) != yes ]] || fatal 'physical write attempted in userspace fastbootd'
    fi
    if [[ $command == flash ]]; then
        image_path=${3:-}
        [[ -f $image_path ]] || fatal "flash image missing for $target"
        image_bytes=$(stat -c %s "$image_path")
        capacity=$(raw_fastboot_value "partition-size:$target")
        [[ $capacity =~ ^0[xX][0-9a-fA-F]+$ || $capacity =~ ^[0-9]+$ ]] || fatal "partition capacity unavailable for $target"
        (( capacity >= image_bytes )) || fatal "$target capacity $capacity is below image size $image_bytes"
    fi
    if ! fb_capture "$label" "$@"; then
        FAILED_WRITE=true
        fatal "first failed write: $label"
    fi
    WRITE_COUNT=$((WRITE_COUNT + 1))
    printf 'WRITE_OK=%s target=%s count=%s\n' "$label" "$target" "$WRITE_COUNT"
}

}

enter_fastbootd_from_android() {
    local label=$1
    [[ $DEVICE_MODE == android ]] || fatal "$label: expected Android before fastbootd entry"
    [[ $("${ADB[@]}" shell getprop ro.boot.slot_suffix | tr -d '\r\n') == _a ]] || fatal "$label: Android slot is not A"
    [[ $(remote_root 'bootctl get-snapshot-merge-status' | tr -d '\r\n') == none ]] || fatal "$label: merge state is not none"
    timeout "$TIMEOUT_SECONDS" "${ADB[@]}" reboot bootloader || fatal "$label: adb reboot bootloader failed"
    DEVICE_MODE=bootloader
    wait_fastboot || fatal "$label: bootloader unavailable"
    [[ $(raw_fastboot_value current-slot) == a ]] || fatal "$label: bootloader slot is not A"
    [[ $(raw_fastboot_value unlocked) == yes ]] || fatal "$label: bootloader is not unlocked"
    fb_capture "$label-reboot-fastboot" reboot fastboot || fatal "$label: reboot fastboot failed"
    wait_fastboot || fatal "$label: userspace fastbootd unavailable"
    DEVICE_MODE=fastbootd
    [[ $(raw_fastboot_value is-userspace) == yes ]] || fatal "$label: is-userspace is not yes"
    [[ $(raw_fastboot_value current-slot) == a ]] || fatal "$label: fastbootd slot is not A"
}

return_a_verify_lpdump() {
    local label=$1
    [[ $DEVICE_MODE == fastbootd ]] || fatal "$label: expected fastbootd before A verification cycle"
    fb_capture "$label-reboot-bootloader" reboot bootloader || fatal "$label: return to bootloader failed"
    wait_fastboot || fatal "$label: bootloader unavailable"
    DEVICE_MODE=bootloader
    [[ $(raw_fastboot_value current-slot) == a ]] || fatal "$label: active slot changed away from A"
    fb_capture "$label-reboot-android" reboot || fatal "$label: reboot A failed"
    DEVICE_MODE=android
    wait_boot_completed || fatal "$label: Android A did not complete boot"
    [[ $("${ADB[@]}" shell getprop ro.boot.slot_suffix | tr -d '\r\n') == _a ]] || fatal "$label: returned Android slot is not A"
    [[ $(remote_root 'bootctl is-slot-bootable 0' | tr -d '\r\n') == 1 ]] || fatal "$label: A is not bootable"
    [[ $(remote_root 'bootctl is-slot-marked-successful 0' | tr -d '\r\n') == 1 ]] || fatal "$label: A is not successful"
    [[ $(remote_root 'bootctl get-snapshot-merge-status' | tr -d '\r\n') == none ]] || fatal "$label: merge state changed"
    remote_root 'for p in boot init_boot vendor_boot dtbo vbmeta vbmeta_system vbmeta_vendor; do sha256sum /dev/block/by-name/${p}_a; done' > "$RUN_DIR/$label.a-bootchain.sha256"
    cmp -s "$RUN_DIR/before.a-bootchain.sha256" "$RUN_DIR/$label.a-bootchain.sha256" || fatal "$label: A physical hashes changed"
    remote_root 'lpdump --slot=0 /dev/block/by-name/super' > "$RUN_DIR/metadata/$label.slot0.txt"
    remote_root 'lpdump --slot=1 /dev/block/by-name/super' > "$RUN_DIR/metadata/$label.slot1.txt"
    verify_lpdumps "$label" || fatal "$label: live lpdump invariant failed"
}
