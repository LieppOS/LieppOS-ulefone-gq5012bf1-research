    echo "ROOT_VBMETA_B_SHA256=$ROOT_SHA"

    [ "$PRE_VENDOR" = "$EXPECTED_PRE_VENDOR" ] || GATE=0
    [ "$PRE_CHILD" = "$EXPECTED_PRE_CHILD" ] || GATE=0
    [ "$SYSTEM_SHA" = "$EXPECTED_SYSTEM" ] || GATE=0
    [ "$VBMETA_SYSTEM_SHA" = "$EXPECTED_VBMETA_SYSTEM" ] || GATE=0
    [ "$ROOT_SHA" = "$EXPECTED_ROOT" ] || GATE=0

    echo "PUSH_CHILD_SHA256=$PUSH_CHILD"

    [ "$PUSH_CHILD_RC" = 0 ] || GATE=0

    [ "$CHILD_WRITE_RC" = 0 ] && \

    echo "POST_VBMETA_VENDOR_B_FIRST4096_SHA256=$POST_CHILD"
    echo "POST_ROOT_VBMETA_B_SHA256=$POST_ROOT"
    echo "POST_VBMETA_SYSTEM_B_FIRST4096_SHA256=$POST_VBMETA_SYSTEM"

    [ "$POST_CHILD" = "$EXPECTED_V20_CHILD" ] && \
    [ "$POST_ROOT" = "$EXPECTED_ROOT" ] && \
    [ "$POST_VBMETA_SYSTEM" = "$EXPECTED_VBMETA_SYSTEM" ] && \

dmctl delete lieppos_vendor_b_rw >/dev/null 2>&1 || true
