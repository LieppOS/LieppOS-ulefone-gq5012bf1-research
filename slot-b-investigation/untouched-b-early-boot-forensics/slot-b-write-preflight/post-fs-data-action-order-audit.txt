# Cgroups are mounted right before early-init using list from /etc/cgroups.json
on early-init
    # Disable sysrq from keyboard
    write /proc/sys/kernel/sysrq 0

    # Android doesn't need kernel module autoloading, and it causes SELinux
    # denials.  So disable it by setting modprobe to the empty string.  Note: to
    # explicitly set a sysctl to an empty string, a trailing newline is needed.
    write /proc/sys/kernel/modprobe \n

    # Set the security context of /adb_keys if present.

===== LATE-INIT EVENT ORDER =====
   552  on late-init
   553      trigger early-fs
   554  
   555      # Mount fstab in init.{$device}.rc by mount_all command. Optional parameter
   556      # '--early' can be specified to skip entries with 'latemount'.
   557      # /system and /vendor must be mounted by the end of the fs stage,
   558      # while /data is optional.
   559      trigger fs
   560      trigger post-fs
   561  
   562      # Mount fstab in init.{$device}.rc by mount_all with '--late' parameter
   563      # to only mount entries with 'latemount'. This is needed if '--early' is
   564      # specified in the previous mount_all command on the fs stage.
   565      # With /system mounted and properties form /system + /factory available,
   566      # some services can be started.
   567      trigger late-fs
   568  
   569      # Now we can mount /data. File encryption requires keymaster to decrypt
   570      # /data, which in turn can only be loaded when system properties are present.
   571      trigger post-fs-data
   572  
   573      # Should be before netd, but after apex, properties and logging is available.
   574      trigger load-bpf-programs
   575      trigger bpf-progs-loaded
   576  
   577      # Now we can start zygote.
   578      trigger zygote-start
   579  
   580      # Remove a file to wake up anything waiting for firmware.
   581      trigger firmware_mounts_complete
   582  
   583      trigger early-boot
   584      trigger boot
   585
